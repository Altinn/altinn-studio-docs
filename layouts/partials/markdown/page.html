{{- /* Markdown-versjonen av en side (index.md), som llms.txt viser AI-verktøy til. Tar imot siden. Vi lager Markdown-en fra den ferdige HTML-en, slik at den har med all tekst leseren ser på siden: tittelen, ingressen og innholdet med kortkodene utført, som tekst fra insert, notice og faner. HTML-kommentarer ser ikke leseren, så dem tar vi ut. */ -}} {{- $html := printf "

%s

" (.Title | htmlEscape) -}} {{- with .Description -}} {{- $html = printf "%s

%s

" $html (. | htmlEscape) -}} {{- end -}} {{- $html = print $html (.Content | replaceRE `(?s)` "") -}} {{- $markdown := $html | transform.HTMLToMarkdown -}} {{- with $markdown }}{{ $markdown = print . "\n" }}{{ end -}} {{- return $markdown -}}
