{{- /* The Markdown version of a page (index.md), which llms.txt points AI tools to. It is made from the rendered HTML, so it has all the text a reader sees on the page: the title, the lead text and the content with every shortcode rendered, such as inserted fragments, notices and tabs. Readers don't see HTML comments, so they are left out. */ -}} {{- $html := printf "

%s

" (.Title | htmlEscape) -}} {{- with .Description -}} {{- $html = printf "%s

%s

" $html (. | htmlEscape) -}} {{- end -}} {{- $html = print $html (.Content | replaceRE `(?s)` "") -}} {{- $markdown := $html | transform.HTMLToMarkdown -}} {{- with $markdown }}{{ $markdown = print . "\n" }}{{ end -}} {{- return $markdown -}}
